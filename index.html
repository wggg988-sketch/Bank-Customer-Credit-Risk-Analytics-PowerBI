DENSE_RANK Window Function (Top Mijozlar) Analytics 
WITH ViloyatReyting AS (
  SELECT mijoz_id, "to'liq_ism", viloyat, kredit_summasi_uzs,
         DENSE_RANK() OVER(
           PARTITION BY viloyat 
           ORDER BY kredit_summasi_uzs DESC
         ) AS orin
  FROM mijozlar WHERE kredit_mavjud = 'Ha'
)
SELECT viloyat, orin, "to'liq_ism", kredit_summasi_uzs
FROM ViloyatReyting WHERE orin <= 3;

IQR Outlier Aniqlash va Churn Xavfi 
# Kredit yuki bo'yicha Outlier aniqlash (IQR)
q1 = df['Kredit_yuklamasi_foiz'].quantile(0.25)
q3 = df['Kredit_yuklamasi_foiz'].quantile(0.75)
iqr = q3 - q1
upper_limit = q3 + 1.5 * iqr

outliers = df[df['Kredit_yuklamasi_foiz'] > upper_limit]
print(f"Outlier mijozlar: {len(outliers)} ta") # 559 ta

# Ketib qolish xavfi (Churn)
df['churn_xavfi'] = np.where(
    (df['oxirgi_kirish_kunlari'] > 60) | (df['faol_mijoz'] == "Yo'q"),
    'Yuqori Churn Xavfi', 'Faol / Sodiq'
)
