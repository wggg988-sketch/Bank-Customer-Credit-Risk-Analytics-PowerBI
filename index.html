02_window_ranking.sql

Har bir viloyat bo‘yicha Top-3 yirik kredit oluvchilar (DENSE_RANK):

WITH ViloyatReyting AS (
  SELECT 
    mijoz_id, "to'liq_ism", viloyat, kredit_turi, kredit_summasi_uzs,
    DENSE_RANK() OVER(
      PARTITION BY viloyat 
      ORDER BY kredit_summasi_uzs DESC
    ) AS orin
  FROM mijozlar
  WHERE kredit_mavjud = 'Ha'
)
SELECT viloyat, orin, mijoz_id, "to'liq_ism", kredit_turi, kredit_summasi_uzs
FROM ViloyatReyting
WHERE orin <= 3
ORDER BY viloyat, orin;
03_credit_risk_query.sql

Kredit muddati kechikkan va skoringi past xavfli mijozlar:

SELECT 
    mijoz_id, "to'liq_ism", viloyat, kredit_reytingi, 
    kredit_summasi_uzs, "muddati_o'tgan_kunlar"
FROM mijozlar
WHERE "muddati_o'tgan_qarz" = 'Ha' 
  AND kredit_reytingi < 600
ORDER BY "muddati_o'tgan_kunlar" DESC
LIMIT 15;
04_portfolio_summary.sql

Umumiy portfel ko‘rsatkichlarini hisoblash so‘rovi:

SELECT 
    COUNT(mijoz_id) AS jami_mijozlar,
    COUNT(CASE WHEN kredit_mavjud = 'Ha' THEN 1 END) AS kredit_olganlar_soni,
    SUM(kredit_summasi_uzs) AS jami_kredit_portfeli,
    SUM(depozit_summasi_uzs) AS jami_depozit_portfeli,
    ROUND(AVG("o'rtacha_oylik_daromad_uzs"), 0) AS ortacha_daromad_uzs,
    ROUND(AVG(kredit_reytingi), 0) AS ortacha_kredit_skori
FROM mijozlar;

01_data_cleaning_and_iqr.py

Kredit yuki bo‘yicha anomal (Outlier) qarz oluvchilarni topish:

# Ustunlar turlarini to'g'irlash va Outlier (IQR) aniqlash
q1 = df['Kredit_yuklamasi_foiz'].quantile(0.25)
q3 = df['Kredit_yuklamasi_foiz'].quantile(0.75)
iqr = q3 - q1
upper_limit = q3 + 1.5 * iqr

outliers_dti = df[df['Kredit_yuklamasi_foiz'] > upper_limit]
print(f"Kredit yuki o'ta yuqori (Outlier) mijozlar soni: {len(outliers_dti)} ta")
