WITH ViloyatReyting AS (
  SELECT 
    mijoz_id, viloyat, "to'liq_ism", kredit_turi, kredit_summasi_uzs,
    DENSE_RANK() OVER(PARTITION BY viloyat ORDER BY kredit_summasi_uzs DESC) AS orin
  FROM mijozlar
  WHERE kredit_mavjud = 'Ha'
)
SELECT viloyat, orin, mijoz_id, "to'liq_ism", kredit_turi, kredit_summasi_uzs
FROM ViloyatReyting
WHERE orin <= 3
ORDER BY viloyat, orin;

Python: Null Qiymatlar va IQR Outlier Skripti Google Colab 
import pandas as pd
import numpy as np

numeric_cols = ["yoshi", "o'rtacha_oylik_daromad_uzs", "kredit_reytingi", 
                "kredit_summasi_uzs", "Kredit_yuklamasi_foiz", "oxirgi_oy_tranzaksiyalar_soni"]

for col in numeric_cols:
    if col in df.columns:
        df[col] = pd.to_numeric(df[col], errors='coerce')

q1 = df['Kredit_yuklamasi_foiz'].quantile(0.25)
q3 = df['Kredit_yuklamasi_foiz'].quantile(0.75)
iqr = q3 - q1
upper_limit = q3 + 1.5 * iqr

outliers_dti = df[df['Kredit_yuklamasi_foiz'] > upper_limit]
print(f"\nKredit yuki o'ta yuqori bo'lgan (Outlier) mijozlar soni: {len(outliers_dti)} ta")
